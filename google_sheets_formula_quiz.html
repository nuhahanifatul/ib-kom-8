Panduan Guru: Setup Google Apps Script

Ikuti langkah-langkah berikut untuk mengintegrasikan skor Evaluasi Praktikum IB Komputer Kelas 8 langsung ke Google Sheets milik Anda:

  1. Buat Spreadsheet baru di Google Drive Anda.
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Hapus seluruh kode default, lalu **Copy & Paste** kode JavaScript di bawah ini:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Header otomatis jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Waktu", "Nama Siswa", "Absen", "Kelas", "Nilai (0-100)", "Benar", "Salah", "Pelanggaran", "Status Ujian"]);
    }
    
    sheet.appendRow([
      data.waktu,
      data.nama,
      data.absen,
      data.kelas,
      data.nilaiAkhir,
      data.skor,
      data.totalSoal - data.skor,
      data.jumlahPelanggaran,
      data.statusSelesai
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
